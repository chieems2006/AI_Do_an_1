def _has_product_noun(norm: str, raw: str) -> bool:
    from msa.textnorm import strip_accents
    low = (raw or "").lower()
    if low and strip_accents(low) != low:   # có dấu
        return bool(_PRODUCT_ACC_C.search(low))
    return bool(_PRODUCT_C.search(norm))
_DEICTIC_C = re.compile(DEICTIC)
_ORDINAL_C = re.compile(ORDINAL)
_SEARCH_VERB_C = re.compile(r"(?<!vua )\btim\b|\bkiem\b|can mua|muon mua|(?<!mua )\bmua\b(?! (mua|he|dong|thu|xuan|nay|lanh|nong))|goi y|de xuat|xem (may|vai|cac)? ?mau|co ban|ban co")
_HELP_C = re.compile(r"tu van|giup (minh|em|toi|tui)|hoi (chut|ti|xiu)|can giup|^(shop|ad|admin) oi$")


_TYPE_WORDS = {
    "footwear": r"giay|dep|sandal|sneaker|boot|cao got|guoc",
    "clothing": r"\bao\b|quan|vay|\bdam\b|khoac|hoodie|so mi",
    "headwear": r"\bmu\b|\bnon\b|luoi trai",
    "accessory": r"\btui\b|balo|that lung|\bkinh\b",
}


def _types_mentioned(norm: str) -> set[str]:
    return {t for t, p in _TYPE_WORDS.items() if re.search(p, norm)}


@dataclass
class RouteResult:
    route: str
    confidence: float
    needs_clarification: bool
    scores: dict[str, float] = field(default_factory=dict)
    reason: str = ""

    def as_dict(self) -> dict:
        return dict(route=self.route, confidence=round(self.confidence, 3), needs_clarification=self.needs_clarification,
                    scores={k: round(v, 2) for k, v in self.scores.items()}, reason=self.reason)


class IntentEmbedder(Protocol):
    def predict(self, text: str) -> tuple[str, float]: ...


class Router:
    """Router(text, image, session) -> {route, confidence, needs_clarification}.

    session (dict, tuỳ chọn): current_product_id, last_results (list product_id), clarify_count (int).
    """

    def __init__(self, tau_intent: float = TAU_INTENT, embedder: IntentEmbedder | None = None, emb_weight: float = 1.0):
        self.tau = tau_intent
        self.embedder = embedder
        self.emb_weight = emb_weight

    # ---------------------------------------------------------- tính điểm
    def score(self, norm: str, session: dict, raw: str = "") -> tuple[dict[str, float], dict[str, Any]]:
        s = {k: 0.0 for k in ROUTES if k not in ("visual", "clarify")}
        for intent, cues in _CUES_C.items():
            for rx, w in cues:
                if rx.search(norm):
                    s[intent] += w
        has_noun = _has_product_noun(norm, raw)
        deictic = bool(_DEICTIC_C.search(norm))
        ordinal = bool(_ORDINAL_C.search(norm))
        has_cur = bool(session.get("current_product_id"))
        n_last = len(session.get("last_results") or [])
        search_verb = bool(_SEARCH_VERB_C.search(norm))
        feats = dict(has_noun=has_noun, search_verb=search_verb, deictic=deictic, ordinal=ordinal, has_cur=has_cur, n_last=n_last, ntok=len(norm.split()))
        # Ngữ cảnh: QA cần 1 SP cụ thể (SP đang xem, hoặc 'đôi thứ 2' trỏ vào kết quả vừa tìm); compare cần >= 2 SP
        ref_one = has_cur or (ordinal and n_last >= 1 and not s["compare"])
        if s["qa"] > 0 and ref_one and (deictic or ordinal or not s["search"]):
            s["qa"] += 1.0
        if ref_one and (deictic or ordinal) and not s["compare"]:
            s["qa"] += 1.0
        if s["qa"] > 0 and not ref_one:
            # hỏi thuộc tính mà không có SP: có danh từ SP -> coi như mô tả để tìm; không có -> cần hỏi lại
            s["qa"] = 0.0 if (has_noun and not deictic) else s["qa"] - 1.0
        # v1: đang ở 1 SP (current_product_id) + câu dạng hỏi, không có động từ tìm/so sánh -> QA về SP đó
        is_question = bool(re.search(r"\b(khong|ko|k|chua|sao|the nao|ntn|bao nhieu|may|gi)\s*$|\?$|\bco\b.+\b(khong|ko|k)\b", norm))
        if has_cur and not search_verb and not s["compare"] and (is_question or s["qa"] > 0):
            s["qa"] += 1.5
            s["search"] *= 0.3  # đang xem 1 SP và hỏi -> không phải tìm mới
        if s["compare"] > 0:
            n_list = len(re.findall(r"\bvs\b|\bva\b|\bvoi\b|\bhay\b", norm)) + raw.count(",") + raw.count(":")
            referents = n_last >= 2 or (n_list >= 1 and has_noun)
            s["compare"] += 1.0 if (referents or (ordinal and n_last >= 2)) else -1.5
            # R19: chỉ so sánh trong cùng product_type – "áo này với đôi giày kia" -> hỏi lại
            if len(_types_mentioned(norm)) >= 2:
                s["compare"] = 0.0
                feats["cross_type_compare"] = True
        if s["gemini_ood"] >= 2.0 and not has_noun:
            s["search"] = 0.0  # "tư vấn mua điện thoại" -> ngoài phạm vi, không phải tìm SP
        if search_verb and not has_noun and not re.search(r"duoi \d|tren \d|\d+ (nghin|trieu)|mau (den|trang|do|xanh)", norm):
            s["search"] = 0.0  # "tìm cái giống vậy", "mua" -> thiếu đối tượng, cần hỏi lại
        if not has_noun and not search_verb:
            s["search"] = 0.0  # bộ lọc (màu/giá/'cho nam') mà không có danh từ SP hay động từ tìm -> không phải search
        if has_noun and (s["search"] > 0) and not (has_cur and deictic):
            s["search"] += 1.0
        if has_noun and s["gemini_ood"] > 0 and (search_verb or (has_cur and deictic)):
            s["gemini_ood"] -= 1.5  # "tìm áo hợp dáng người gầy" là search; "áo này giặt sao" là qa
        if has_noun and s["gemini_chitchat"] > 0 and (s["search"] > 0 or s["qa"] > 0):
            s["gemini_chitchat"] -= 1.0
        return s, feats

    def route(self, text: str | None, image: bytes | None = None, session: dict | None = None) -> dict:
        return self.route_obj(text, image, session).as_dict()

    def route_obj(self, text: str | None, image: bytes | None = None, session: dict | None = None) -> RouteResult:
        session = session or {}
        norm = normalize(text)
        if image:
            # Có ảnh => luôn qua visual trước (Gemini không nhận ảnh – mục 5.4); câu kèm theo dùng làm bộ lọc.
            return RouteResult("visual", 0.95 if len(norm.split()) <= 12 else 0.85, False, {}, "co_anh")
        if not norm:
            return self._low("clarify", 0.0, {}, session, "rong")
        s, f = self.score(norm, session, text or "")
        if self.embedder is not None:
            lab, sim = self.embedder.predict(text or "")
            if lab in s:
                s[lab] += self.emb_weight * max(sim, 0.0)
        if f.get("cross_type_compare"):
            return self._low("compare", 0.3, s, session, "so_sanh_khac_nhom")
        ranked = sorted(s.items(), key=lambda kv: kv[1], reverse=True)
        (best, b), (_, b2) = ranked[0], ranked[1]
        if b <= 0:
            # không khớp gì: câu ngắn -> hỏi lại; có danh từ SP -> search; còn lại -> Gemini ood
            if f["has_noun"] and f["ntok"] >= 2 and not (f["ordinal"] and f["n_last"] < 2):
                return RouteResult("search", 0.55, False, s, "chi_co_danh_tu_sp")
            if f["ntok"] <= 4 or f["deictic"] or f["ordinal"] or _HELP_C.search(norm):
                return self._low("gemini_ood", 0.3, s, session, "mo_ho")
            return RouteResult("gemini_ood", 0.5, False, s, "khong_khop_luat")
        conf = b / (b + max(b2, 0.0) + 0.5)
        if conf < self.tau:
            return self._low(best, conf, s, session, "duoi_tau")
        return RouteResult(best, conf, False, s, "luat")

    def _low(self, best: str, conf: float, s: dict, session: dict, why: str) -> RouteResult:
        """Không chắc: hỏi lại 1 lần; nếu đã hỏi lại rồi thì chuyển Gemini (hoặc giữ route nội bộ nếu là best)."""
        if int(session.get("clarify_count", 0)) >= 1:
            target = best if best.startswith("gemini_") else "gemini_ood"
            return RouteResult(target, conf, False, s, why + "+da_hoi_lai")
        return RouteResult("clarify", conf, True, s, why)

@dataclass
class Tools:
    """Công cụ nội bộ (tiêm phụ thuộc để ghép code của Khoa/Chiến/Hoàng)."""
    visual: Callable[..., list[dict]] | None = None          # VisualSearcher.search(image, k, product_type)
    qa: Callable[..., dict] | None = None                    # RAGPipeline.answer(question, product_id)
    search: Callable[..., list[dict]] | None = None          # search_text(query, filters) -> items
    compare: Callable[..., dict] | None = None               # compare(product_ids) -> {answer, items}


class Agent:
    """Agent.handle: route -> công cụ nội bộ trước -> Gemini khi chitchat/ood/điểm < τ. Trả định dạng response chung (mục 5.2)."""

    def __init__(self, router: Router | None = None, tools: Tools | None = None, llm: LLMGateway | None = None,
                 tau_v: float = TAU_V, tau_r: float = TAU_R, postcheck: bool = True):
        self.router = router or Router()
        self.tools = tools or Tools()
        self.llm = llm
        self.tau_v, self.tau_r, self.postcheck = tau_v, tau_r, postcheck

    def handle(self, text: str | None = None, image: bytes | None = None, session: dict | None = None) -> dict:
        t0 = time.perf_counter()
        session = session if session is not None else {}
        rr = self.router.route_obj(text, image, session)
        resp: dict[str, Any] = dict(route=rr.route, confidence=round(rr.confidence, 3), source="internal", answer="",
                                    items=[], citations=[], latency_ms=0, request_id=str(uuid.uuid4()))
        if rr.route == "clarify":
            session["clarify_count"] = int(session.get("clarify_count", 0)) + 1
            resp["answer"] = CLARIFY_TEXT
        elif rr.route.startswith("gemini_"):
            self._gemini(text or "", resp)
        else:
            session["clarify_count"] = 0
            self._internal(rr.route, text or "", image, session, resp)
        resp["latency_ms"] = round((time.perf_counter() - t0) * 1000, 1)
        return resp

    # ------------------------------------------------------------------
    def _internal(self, route: str, text: str, image: bytes | None, session: dict, resp: dict) -> None:
        last = list(session.get("last_results") or [])
        if route == "visual" and self.tools.visual:
            items = self.tools.visual(image, k=10)
            if not items or items[0].get("score", 0) < self.tau_v:
                resp["answer"] = "Mình chưa thấy sản phẩm đủ giống trong cửa hàng. Bạn mô tả bằng chữ (loại, màu, kiểu) giúp mình nhé."
            else:
                resp["items"] = items
                resp["answer"] = f"Mình tìm được {len(items)} sản phẩm giống ảnh."
                session["last_results"] = [i["product_id"] for i in items]
        elif route == "search" and self.tools.search:
            items = self.tools.search(text)
            resp["items"] = items
            resp["answer"] = f"Có {len(items)} sản phẩm phù hợp." if items else "Mình chưa tìm thấy sản phẩm phù hợp."
            if items:
                session["last_results"] = [i["product_id"] for i in items]
        elif route == "qa" and self.tools.qa:
            pid = resolve_product_ref(text, session) or session.get("current_product_id")
            out = self.tools.qa(text, pid)
            resp["answer"], resp["citations"] = out.get("answer", ""), out.get("citations", [])
            if out.get("refused") or out.get("retrieval_score", 1.0) < self.tau_r:
                # Quy tắc 5.4: thiếu dữ kiện SP thì giữ "chưa có thông tin" – Gemini KHÔNG bù.
                resp["answer"] = out.get("answer") or "Mình chưa có thông tin này cho sản phẩm bạn đang xem."
        elif route == "compare" and self.tools.compare:
            ids = resolve_compare_refs(text, session) or last[:2]
            out = self.tools.compare(ids)
            resp["answer"], resp["items"] = out.get("answer", ""), out.get("items", [])
        else:
            resp["answer"] = f"(Công cụ '{route}' chưa được nối.)"

    def _gemini(self, text: str, resp: dict) -> None:
        if self.llm is None:
            resp.update(source="internal", answer="Hiện mình chỉ hỗ trợ tìm và hỏi đáp sản phẩm thời trang.")
            return
        out = self.llm.generate(text, system=GEMINI_SYSTEM_PROMPT)
        ans = out.get("answer") or ""
        if out.get("error"):
            resp.update(source="internal", answer="Xin lỗi, phần trò chuyện đang bận. Bạn thử lại sau hoặc tìm sản phẩm giúp mình nhé.")
            resp["error"] = out["error"]
            return
        facts = find_store_facts(ans) if self.postcheck else []
        if facts:
            ans = SAFE_REPLACEMENT
        resp.update(source="cache" if out.get("cached") else "gemini", answer=ans, store_fact_flags=facts,
                    tokens_in=out.get("tokens_in"), tokens_out=out.get("tokens_out"))
